  18  class DiagonalParameterDefinition(nn.Module):
  19      def __init__(self, width):
  20          super().__init__()
  21          self.wq = nn.Parameter(torch.ones(width))
  22          self.wk = nn.Parameter(torch.ones(width))
  23          self.wv = nn.Parameter(torch.ones(width))
  24          # softplus(inverse_softplus(1)) = 1 initially.
  25          self.raw_pair_scale = nn.Parameter(torch.full((3,), math.log(math.expm1(1.))))
  26  
  27      def pair_scales(self):
  28          return F.softplus(self.raw_pair_scale).clamp_min(1e-6)

  31  class DiagonalRelationBlock(nn.Module):
  32      def __init__(self, width):
  33          super().__init__()
  34          self.p = DiagonalParameterDefinition(width)
  35          self.norm = nn.LayerNorm(width)
  36          self.ffn = nn.Sequential(
  37              nn.LayerNorm(width), nn.Linear(width, 2 * width),
  38              nn.GELU(), nn.Linear(2 * width, width),
  39          )
  40  
  41      def forward(self, h, batch, return_relations=False):
  42          # h [B,N,D]; all node-count-dependent tensors come from graph structure.
  43          # Explorer STEP 06 / 11: 两层分别执行节点归一化，各层参数独立。
  44          # LaTeX: $$Z=\operatorname{LayerNorm}(H)$$
  45          z = self.norm(h)
  46          # Explorer STEP 07 / 12: 按通道逐元素缩放；V 在关系矩阵计算后求值。
  47          # LaTeX: $$Q=Z\odot w_q,\quad K=Z\odot w_k,\quad V=Z\odot w_v$$
  48          q, k = z * self.p.wq, z * self.p.wk
  49          # Explorer STEP 08 / 13: 结构类型缩放与祖先掩码；不可达位置设为负无穷。补齐行保留自身。
  50          # LaTeX: $$T=QK^{\top}/\sqrt{d},\quad C_{ij}=\max(\operatorname{softplus}(r_{type(i,j)}),10^{-6}),\quad A=\operatorname{softmax}_{j}(\operatorname{mask}_{M}(T\odot C))$$
  51          similarity = q @ k.transpose(-1, -2) / math.sqrt(h.shape[-1])
  52          n = h.shape[1]
  53          eye = torch.eye(n, dtype=torch.bool, device=h.device)[None]
  54          kind = torch.where(batch['distances'] == 1, 1, 2)
  55          kind = torch.where(eye, 0, kind)
  56          scales = self.p.pair_scales()
  57          pair_scale = scales[kind]
  58          weighted = similarity * pair_scale
  59          # Padding rows retain a self entry to avoid all-masked softmax; padding
  60          # cannot send messages to valid nodes and is excluded from readout.
  61          allowed = batch['allowed'] & batch['valid'][:, None, :]
  62          allowed = allowed | (eye & ~batch['valid'][:, :, None])
  63          logits = weighted.masked_fill(~allowed, -torch.inf)
  64          relation = logits.softmax(dim=-1)
  65          v = z * self.p.wv
  66          # Explorer STEP 09 / 14: 聚合消息并加上本层原始输入的残差。
  67          # LaTeX: $$U=H+AV$$
  68          message = relation @ v
  69          h = h + message
  70          # Explorer STEP 10 / 15: 先归一化，再扩展通道、GELU、回投影，最后加回 U。
  71          # LaTeX: $$H^{\prime}=U+\operatorname{GELU}(\operatorname{LayerNorm}(U)W_1^{\top}+b_1)W_2^{\top}+b_2$$
  72          h = h + self.ffn(h)
  73          debug = None
  74          if return_relations:
  75              debug = {'similarity': similarity, 'pair_scale': pair_scale,
  76                       'weighted_similarity': weighted, 'relation': relation,
  77                       'allowed': allowed, 'relation_type': kind,
  78                       'value': v, 'message': message,
  79                       'wq': self.p.wq, 'wk': self.p.wk, 'wv': self.p.wv,
  80                       'qk_channel_product': self.p.wq * self.p.wk,
  81                       'type_scales': scales}
  82          return h, debug

  85  class DiagonalGraphScoreModel(nn.Module):
  86      def __init__(self, width=32, layers=2, tasks=1):
  87          super().__init__()
  88          self.config = dict(width=width, layers=layers, tasks=tasks)
  89          self.input = nn.Linear(len(FEATURE_COLUMNS), width)
  90          self.blocks = nn.ModuleList([DiagonalRelationBlock(width) for _ in range(layers)])
  91          # Match V1's pooling and readout so the comparison changes the blocks.
  92          self.task = nn.Embedding(tasks, 16)
  93          self.readout = nn.Sequential(nn.Linear(2 * width + 17, width), nn.GELU(),
  94                                       nn.Linear(width, width), nn.GELU())
  95          self.score = nn.Linear(width, 1)
  96          self.completion = nn.Linear(width, 1)
  97  
  98      def forward(self, batch, return_relations=False):
  99          h = self.input(batch['x'])
 100          relations = []
 101          for block in self.blocks:
 102              h, debug = block(h, batch, return_relations)
 103              if debug is not None:
 104                  relations.append(debug)
 105          valid = batch['valid'].unsqueeze(-1)
 106          count = valid.sum(1).clamp_min(1)
 107          total = (h * valid).sum(1)
 108          features = torch.cat((total / count, total / 32, self.task(batch['task']),
 109                                count.float().log1p() / 5), dim=-1)
 110          pooled = self.readout(features)
 111          result = {'score': self.score(pooled).squeeze(-1),
 112                    'completion_logit': self.completion(pooled).squeeze(-1)}
 113          if return_relations:
 114              result['relations'] = relations
 115              result['node_states'] = h
 116          return result
