   1  """Label-free graph features; control scopes are connected by explicit boundary edges."""
   2  from dataclasses import dataclass
   3  import math
   4  import torch
   5  from ..graph_ir.analysis import FEATURE_TYPES
   6  from ..graph_ir.validation import split_ref

   8  PORTS=('x','other','weight','weights','q','k','scores','mask','logits','values','xs','state','fragment','carry','y','out','capture','binding','body_output','self')
   9  ACTIVATIONS=('relu','gelu','silu','sigmoid','tanh','identity')
  10  REDUCERS=('sum','mean','max')
  11  NUMERIC=('rank','numel','dim0','dim1','dim2','dim3','stored_depth','scope_depth','steps','shared','per_step','axis','offset','trainable','parameter_numel')
  12  FEATURE_COLUMNS=tuple(FEATURE_TYPES)+NUMERIC+tuple('activation:'+s for s in ACTIVATIONS)+tuple('reducer:'+s for s in REDUCERS)

  25  def encode(graph,batch_size=64):
  26      # Explorer STEP 01: 确定性编码：节点属性与有向图关系。
  27      # LaTeX: $$G \mapsto (X_{raw},E_{raw},M_{raw},D_{raw}),\quad X_{raw}\in\mathbb{R}^{N\times46}$$
  28      exported=graph.matrices({'B':batch_size})
  29      scopes=exported['scopes'];keys=[];rows=[];mapping={};stats=[0.]*len(FEATURE_TYPES)
  30      for scope,data in sorted(scopes.items()):
  31          definitions={n['id']:n for n in data['nodes']}
  32          for i,(name,kind) in enumerate(zip(data['node_ids'],data['node_types'])):
  33              mapping[(scope,name)]=len(rows);keys.append(scope+'/'+name)
  34              attrs=definitions.get(name,{}).get('attrs',{})
  35              specs=[v for k,v in data['tensor_specs'].items() if k.startswith(name+':')]
  36              if not specs:
  37                  specs=[e['tensor'] for e in data['edges'] if e['target']==name]
  38              shape=specs[0]['shape'] if specs else []
  39              dims=[math.log1p(v)/10 for v in shape[:4]]+[0.]*(4-len(shape[:4]))
  40              param=data['parameters'].get(name,{})
  41              pnum=math.prod(param['shape']) if param else 0
  42              numeric=[len(shape)/4,math.log1p(math.prod(shape))/10,*dims,
  43                       data['stored_depth'][i]/20,scope.count('/')/4,math.log1p(attrs.get('steps',0))/4,
  44                       float(attrs.get('binding')=='shared'),float(attrs.get('binding')=='per_step'),
  45                       float(attrs.get('dim',0)) if isinstance(attrs.get('dim',0),int) else 0.,float(attrs.get('offset',0)),
  46                       float(param.get('trainable',False)),math.log1p(pnum)/10]
  47              row=data['node_features'][i]+numeric+[float(attrs.get('activation')==a) for a in ACTIVATIONS]+[float(attrs.get('reducer')==a) for a in REDUCERS]
  48              rows.append(row);stats[FEATURE_TYPES.index(kind)]+=1
  49      n=len(rows);relations=torch.zeros(n,n,len(PORTS));direct=torch.zeros(n,n,dtype=torch.bool)
  50      def add(source,target,port):
  51          direct[target,source]=True
  52          relations[target,source,PORTS.index(port if port in PORTS else 'capture')]+=1
  53      for scope,data in scopes.items():
  54          for edge in data['edges']:add(mapping[(scope,edge['source'])],mapping[(scope,edge['target'])],edge['target_port'])
  55      for relation in exported['hierarchy']:
  56          parent=relation['parent_scope'];child=relation['body_scope'];control=relation['control_node']
  57          # Hierarchical input bindings preserve which external tensor is read by the body.
  58          for port,refs in relation['bindings'].items():
  59              target_port='fragment' if port=='xs' else port
  60              if (child,target_port) not in mapping:continue
  61              for ref in refs if isinstance(refs,list) else [refs]:
  62                  name,_=split_ref(ref)
  63                  add(mapping[(parent,name)],mapping[(child,target_port)],'binding')
  64          for name,kind in zip(scopes[child]['node_ids'],scopes[child]['node_types']):
  65              if kind=='OUTPUT':add(mapping[(child,name)],mapping[(parent,control)],'body_output')
  66      distance=torch.full((n,n),n+1,dtype=torch.long);distance[direct]=1;distance.fill_diagonal_(0)
  67      for k in range(n):distance=torch.minimum(distance,distance[:,k,None]+distance[k,None,:])
  68      allowed=distance<=n
  69      relations[torch.arange(n),torch.arange(n),PORTS.index('self')]=1
  70      distances=distance.clamp(max=8)
  71      pnum=sum(math.prod(p.shape) for p in graph.parameters.values() if p.trainable)
  72      summary=[math.log1p(v) for v in stats]+[math.log1p(pnum),math.log1p(n),float(max(d['stored_depth'][-1] for d in scopes.values()))/20]
  73      return EncodedGraph(torch.tensor(rows,dtype=torch.float32),allowed,relations,distances,torch.tensor(summary,dtype=torch.float32),keys)

  76  def batch_graphs(graphs,task_ids=None):
  77      """Pad graphs; omit task_ids for one-model-per-task predictors.
  78  
  79      Explicit task_ids retain the legacy task field for older predictors.
  80      """
  81      if not graphs or (task_ids is not None and len(graphs)!=len(task_ids)):
  82          raise ValueError('Need nonempty graphs and matching task_ids when supplied')
  83      size=max(len(g.ids) for g in graphs);b=len(graphs)
  84      x=torch.zeros(b,size,len(FEATURE_COLUMNS));valid=torch.zeros(b,size,dtype=torch.bool)
  85      mask=torch.eye(size,dtype=torch.bool).expand(b,-1,-1).clone()
  86      relations=torch.zeros(b,size,size,len(PORTS));dist=torch.full((b,size,size),8,dtype=torch.long)
  87      for i,g in enumerate(graphs):
  88          n=len(g.ids);x[i,:n]=g.x;valid[i,:n]=True;mask[i,:n,:n]=g.allowed
  89          relations[i,:n,:n]=g.relations;dist[i,:n,:n]=g.distances
  90      batch={'x':x,'valid':valid,'allowed':mask,'relations':relations,'distances':dist,
  91             'statistics':torch.stack([g.statistics for g in graphs])}
  92      if task_ids is not None:
  93          batch['task']=torch.tensor(task_ids,dtype=torch.long)
  94      return batch
