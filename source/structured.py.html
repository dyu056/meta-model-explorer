  20  class StructureParameterDefinition(nn.Module):
  21      """Own structural gains and translate graph metadata into aligned matrices.
  22  
  23      Input is the batch_graphs dictionary. x: [B,N,F], valid: [B,N],
  24      allowed/distances: [B,N,N], relations: [B,N,N,R]. ``order[b,k]`` is
  25      the original index of sorted node k; inverse_order undoes that permutation.
  26      Neither sorting nor fixed dependency matrices have trainable parameters.
  27      """
  28  
  29      def __init__(self, sparse: bool = True):
  30          super().__init__()
  31          self.sparse = sparse
  32          if sparse:
  33              self.self_scale = nn.Parameter(torch.tensor(1.0))
  34              self.predecessor_scale = nn.Parameter(torch.tensor(0.0))
  35          else:
  36              self.register_buffer('self_scale', torch.tensor(1.0))
  37              self.register_buffer('predecessor_scale', torch.tensor(0.0))
  38  
  39      @staticmethod
  40      def direct_dependencies(batch):
  41          """Boolean [B,N,N]; shortest distance 1 identifies a direct edge."""
  42          valid = batch['valid']
  43          n = valid.shape[1]
  44          eye = torch.eye(n, dtype=torch.bool, device=valid.device)[None]
  45          return ((batch['distances'] == 1) & batch['allowed'] & ~eye
  46                  & valid[:, :, None] & valid[:, None, :])
  47  
  48      @staticmethod
  49      def reorder_nodes(value, order):
  50          """Apply each graph's own permutation along node axis 1."""
  51          shape = (*order.shape, *((1,) * (value.ndim - 2)))
  52          return value.gather(1, order.reshape(shape).expand_as(value))
  53  
  54      @classmethod
  55      def reorder_pairs(cls, value, order):
  56          """Synchronously permute receiver and source axes, preserving ports."""
  57          value = cls.reorder_nodes(value, order)
  58          shape = (order.shape[0], 1, order.shape[1], *((1,) * (value.ndim - 3)))
  59          return value.gather(2, order.reshape(shape).expand_as(value))
  60  
  61      @classmethod
  62      def prepare(cls, batch):
  63          """Kahn topological sort; reject cycles, place padding last.
  64  
  65          CPU traversal is metadata preparation, not a differentiable operation.
  66          Independent branches may tie; their index tie-break has no effect on
  67          predictions because no node-index/bandwidth parameters are learned.
  68          """
  69          valid = batch['valid']
  70          if valid.ndim != 2 or valid.shape[1] == 0:
  71              raise ValueError('Expected nonempty padded node axis [B,N]')
  72          direct = cls.direct_dependencies(batch)
  73          orders = []
  74          for graph, mask in zip(direct.detach().cpu(), valid.detach().cpu()):
  75              nodes = mask.nonzero(as_tuple=True)[0].tolist()
  76              if not nodes:
  77                  raise ValueError('Each graph needs at least one valid node')
  78              degree = graph.sum(-1).tolist()
  79              ready = [i for i in nodes if degree[i] == 0]
  80              heapq.heapify(ready)
  81              order = []
  82              successors = graph.transpose(0, 1)
  83              while ready:
  84                  source = heapq.heappop(ready)
  85                  order.append(source)
  86                  for target in successors[source].nonzero(as_tuple=True)[0].tolist():
  87                      degree[target] -= 1
  88                      if degree[target] == 0:
  89                          heapq.heappush(ready, target)
  90              if len(order) != len(nodes):
  91                  raise ValueError('Topological encoding requires an acyclic dependency graph')
  92              orders.append(order + (~mask).nonzero(as_tuple=True)[0].tolist())
  93          order = torch.tensor(orders, device=valid.device, dtype=torch.long)
  94          # Explorer STEP 02: 拓扑排序后，节点轴与成对矩阵的两轴同步重排。
  95          # LaTeX: $$X=\Pi X_{raw},\quad B=\Pi B_{raw}\Pi^\top\ (B\in\{E,M,D\})$$
  96          aligned = dict(batch)
  97          for key in ('x', 'valid'):
  98              aligned[key] = cls.reorder_nodes(batch[key], order)
  99          for key in ('allowed', 'distances', 'relations'):
 100              aligned[key] = cls.reorder_pairs(batch[key], order)
 101          return aligned, order, order.argsort(dim=1)

 103      @classmethod
 104      def fixed_matrices(cls, batch, dtype):
 105          """I on valid nodes and P = row-normalized direct predecessors.
 106  
 107          P has a zero diagonal; roots have a zero row. Residual addition supplies
 108          self information exactly once during the fixed aggregation stage.
 109          """
 110          valid = batch['valid']
 111          direct = cls.direct_dependencies(batch).to(dtype)
 112          predecessors = direct / direct.sum(-1, keepdim=True).clamp_min(1)
 113          identity = torch.diag_embed(valid.to(dtype))
 114          return identity, predecessors
 115  
 116      def matrices(self, batch, dtype):
 117          identity, predecessors = self.fixed_matrices(batch, dtype)
 118          sparse = self.self_scale.to(dtype) * identity + self.predecessor_scale.to(dtype) * predecessors
 119          return sparse, predecessors

 122  class SparseStructureEncoder(nn.Module):
 123      """[B,N,D] -> [B,N,D], with S = a I + b P, then (I + P) S H.
 124  
 125      a=1,b=0 initially. Gains may become signed. S is structurally sparse but is
 126      materialized as a dense batched tensor for the current small-graph backend.
 127      """
 128  
 129      def __init__(self, sparse: bool = True):
 130          super().__init__()
 131          self.p = StructureParameterDefinition(sparse)
 132  
 133      def forward(self, h, sparse_matrix, predecessors):
 134          # Explorer STEP 04: 结构变换：P 为归一化直接前驱，a、b 为可学习标量。
 135          # LaTeX: $$P_{ij}=E_{ij}/\max(1,\sum_k E_{ik}),\quad S=aI+bP,\quad L=SH_0$$
 136          local = dsl.mix(h, sparse_matrix)
 137          # Explorer STEP 05: 固定前驱聚合，加回自身表示。
 138          # LaTeX: $$H_s=L+PL$$
 139          encoded = dsl.add(local, dsl.mix(local, predecessors))
 140          return encoded, local

 143  class StructuredGraphScoreModel(nn.Module):
 144      """Topology -> projection -> sparse/fixed encoding -> diagonal relation blocks.
 145  
 146      Defaults: feature width 32, two independent relation blocks. Train a separate
 147      model per task; there is no task embedding or required task ID in the batch.
 148      ``objective='normalized_score'`` returns one sigmoid score [B] and owns no
 149      completion head. The default ``legacy`` mode retains the historical raw
 150      score/completion_logit outputs for existing relation-v2 experiments.
 151      With return_relations=True, return
 152      sorted-node states, order/inverse_order, structural matrices/states, and
 153      each block's raw similarity, weighted similarity, softmax relation and V.
 154      Use node_order to align node IDs; use inverse_order to restore tensors.
 155      """
 156  
 157      def __init__(self, width=32, layers=2, sparse=True, objective='legacy'):
 158          super().__init__()
 159          if objective not in ('legacy', 'normalized_score'):
 160              raise ValueError('objective must be legacy or normalized_score')
 161          self.config = dict(width=width, layers=layers, sparse=sparse, objective=objective)
 162          self.objective = objective
 163          self.input = nn.Linear(len(FEATURE_COLUMNS), width)
 164          self.blocks = nn.ModuleList([DiagonalRelationBlock(width) for _ in range(layers)])
 165          # Masked mean, fixed-scale sum, and log node count: 2 * width + 1.
 166          self.readout = nn.Sequential(nn.Linear(2 * width + 1, width), nn.GELU(),
 167                                       nn.Linear(width, width), nn.GELU())
 168          self.score = nn.Linear(width, 1)
 169          if objective == 'legacy':
 170              self.completion = nn.Linear(width, 1)
 171          self.encoder = SparseStructureEncoder(sparse=sparse)
 172  
 173      def forward(self, batch, return_relations=False):
 174          aligned, order, inverse = self.encoder.p.prepare(batch)
 175          # Explorer STEP 03: 共享线性投影：每个节点从 46 维映射为 32 维。
 176          # LaTeX: $$H_0=XW_{in}^{\top}+b_{in}$$
 177          h = self.input(aligned['x'])
 178          sparse, predecessors = self.encoder.p.matrices(aligned, h.dtype)
 179          projected = h
 180          h, local = self.encoder(h, sparse, predecessors)
 181          structural = h
 182          relations = []
 183          for block in self.blocks:
 184              h, debug = block(h, aligned, return_relations)
 185              if debug is not None:
 186                  relations.append(debug)
 187          # Explorer STEP 16: 只汇聚有效节点，拼接均值、固定缩放求和与节点数量。
 188          # LaTeX: $$n=\sum_i v_i,\quad t=\sum_i v_i H_i,\quad z=[t/\max(n,1);\ t/32;\ \ln(1+\max(n,1))/5]$$
 189          valid = aligned['valid'].unsqueeze(-1)
 190          count = valid.sum(1).clamp_min(1)
 191          total = (h * valid).sum(1)
 192          features = torch.cat((total / count, total / 32,
 193                                count.float().log1p() / 5), dim=-1)
 194          # Explorer STEP 17: 图级 MLP：65 → 32 → 32。
 195          # LaTeX: $$g=\operatorname{GELU}(\operatorname{GELU}(zW_1^{\top}+b_1)W_2^{\top}+b_2)$$
 196          pooled = self.readout(features)
 197          # Explorer STEP 18: normalized_score 模式返回单一 sigmoid 分数；标签不参与前向。
 198          # LaTeX: $$\ell=gw+b,\quad \widehat{s}=\sigma(\ell)$$
 199          score = self.score(pooled).squeeze(-1)
 200          if self.objective == 'normalized_score':
 201              result = {'score': score.sigmoid()}
 202          else:
 203              result = {'score': score,
 204                        'completion_logit': self.completion(pooled).squeeze(-1)}
 205          if return_relations:
 206              result.update(node_order=order, inverse_order=inverse,
 207                            node_states=h, relations=relations,
 208                            structure={'sparse_matrix': sparse, 'predecessors': predecessors,
 209                                       'projected': projected, 'after_sparse': local,
 210                                       'after_dependency': structural,
 211                                       'self_scale': self.encoder.p.self_scale,
 212                                       'predecessor_scale': self.encoder.p.predecessor_scale,
 213                                       'valid': aligned['valid']})
 214          return result
